# --- Object detection (YOLOv11) -------------------------------------------
# Ultralytics pulls in torch/torchvision. Optional at runtime: the pipeline
# degrades gracefully if these are unavailable.
#
# PyPI's default Linux torch wheel bundles the full CUDA toolkit as required
# dependencies (~2GB+) even when no GPU is present — this container has none,
# and YOLO_DEVICE=auto (see app/services/object_detection.py) already falls
# back to CPU inference when no GPU is available, so that download buys
# nothing here. Pinning the explicit +cpu build from PyTorch's own CPU wheel
# index (~175MB total) gets identical CPU-inference behavior for a fraction
# of the image size and build time. If you need GPU acceleration, remove the
# extra-index-url line and these two pins so ultralytics' normal torch>=1.8.0
# range resolves the default (CUDA-enabled) PyPI wheel instead.
--extra-index-url https://download.pytorch.org/whl/cpu
torch==2.5.1+cpu
torchvision==0.20.1+cpu
ultralytics==8.3.58

# --- Authentication ----------------------------------------------------------
# Plain `bcrypt` rather than passlib: passlib 1.7.x is unmaintained and
# incompatible with bcrypt>=4.1 (raises on hash/verify — see
# https://github.com/pyca/bcrypt/issues/684). Using bcrypt's own API directly
# avoids the dependency entirely.
#
# JWT is hand-rolled (HS256 only, stdlib hmac/hashlib) in app/core/security.py
# rather than depending on PyJWT: this app only ever needs symmetric HS256
# signing, and PyJWT probes for the optional `cryptography` package (for
# RSA/EC algorithms this app never uses) at import time — on a system with a
# broken/mismatched `cryptography` install (a compiled Rust extension) that
# probe can crash before any of this code runs, for a dependency this app
# doesn't actually need. See the module docstring for the full rationale.
bcrypt==4.2.1
